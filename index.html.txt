<div id="word" style="font-size:50px;">CLICK</div>

<script>
    var words = [
        "HELLO",
        "WAIT",
        "ONE",
        "MORE",
        "CLICK",
        "KEEP",
        "GOING",
        "ALMOST",
        "THERE",
        "DONE"
    ];

    var i = 0;

    document.body.onclick = function () {
        i = i + 1;

        if (i < words.length) {
            document.getElementById("word").innerHTML = words[i];
        } else {
            document.getElementById("word").innerHTML = "THE END";
        }
    };
</script>
